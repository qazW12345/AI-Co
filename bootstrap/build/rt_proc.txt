# rt_proc.c has no dependency on the WP-M0-06 diag module: rt.proc
# emits no diagnostic records (args() has no failure mode; exit() is a
# plain termination, not a trap), so it needs no diag sources. It
# depends only on the Windows API (GetCommandLineW, kernel32) and the
# C library (exit), and on its own header.
#
